# launch_server 概览

launch_server 是 mini-sglang 的启动入口，如下，它会开启若干个线程，包括
* world_size 个 _run_scheduler 线程
* 1 个 DeTokenizer ，它负责将模型吐出的 token 转化为文本
* num_tokenizer 个 tokenzier， 负责将文本转化为token。tokenizer 和 Detokenizer 之间的通信是通过 zmq 来做的

```python
def launch_server(run_shell: bool = False) -> None:

    server_args, run_shell = parse_args(sys.argv[1:], run_shell)
    logger = init_logger(__name__, "initializer")

    def start_subprocess() -> None:

        world_size = server_args.tp_info.size
        # a multiprocessing queue to receive ack from subprocesses
        # so that we can guarantee all subprocesses are ready
        ack_queue: mp.Queue[str] = mp.Queue()

        for i in range(world_size):
            mp.Process(
                target=_run_scheduler,
                args=(new_args, ack_queue),
                daemon=False,
                name=f"minisgl-TP{i}-scheduler",
            ).start()

        num_tokenizers = server_args.num_tokenizer
        # DeTokenizer, only 1
        mp.Process(
            target=tokenize_worker,
            kwargs={
                # ...
            },
            daemon=False,
            name="minisgl-detokenizer-0",
        ).start()
        for i in range(num_tokenizers):
            mp.Process(
                target=tokenize_worker,
                kwargs={
                    # ...
                },
                daemon=False,
                name=f"minisgl-tokenizer-{i}",
            ).start()

        # Wait for acknowledgments from all worker processes:
        # - world_size schedulers (but only primary rank sends ack)
        # - num_tokenizers tokenizers
        # - 1 detokenizer
        # Total acks expected: 1 + num_tokenizers + 1 = num_tokenizers + 2
        for _ in range(num_tokenizers + 2):
            logger.info(ack_queue.get())
```

tokenizer和Detokenizer的内容已经看过了，重点看下 __run_scheduler 吧:

```python
def _run_scheduler(args: ServerArgs, ack_queue: mp.Queue[str]) -> None:

    with torch.inference_mode():
        scheduler = Scheduler(args)
        scheduler.sync_all_ranks()

        if args.tp_info.is_primary():
            ack_queue.put("Scheduler is ready")

        try:
            scheduler.run_forever()
        except KeyboardInterrupt:
            ...
           scheduler.shutdown()
```

显然核心逻辑是 `run_forever`，而 `run_forever` 有两条分支，上面对应的是串行调度，死循环调用 `normal_loop`,
下面对应的是流水线重叠，死循环调用 `overlap_loop`, 为简单起见，我们先研读串行调度的代码。
```python
class Scheduler(SchedulerIOMixin):
    @torch.inference_mode()
    def run_forever(self) -> NoReturn:
        if ENV.DISABLE_OVERLAP_SCHEDULING:
            with self.engine_stream_ctx:
                self.engine.stream.wait_stream(self.stream)
                while True:
                    self.normal_loop()
        else:
            assert torch.cuda.current_stream() == self.stream
            data = None
            while True:
                data = self.overlap_loop(data)
```

# Scheduler.normal_loop() 详解

## Scheduler.normal_loop() 源码初探
```python
class Scheduler(SchedulerIOMixin):
    def normal_loop(self) -> None:
        blocking = not (self.prefill_manager.runnable or self.decode_manager.runnable)
        for msg in self.receive_msg(blocking=blocking):
            self._process_one_msg(msg)

        forward_input = self._schedule_next_batch()
        ongoing_data = None
        if forward_input is not None:
            ongoing_data = (forward_input, self._forward(forward_input))

        self._process_last_data(ongoing_data)
```
第一行的 blocking 的含义是如果当前没有 prefill 任务可跑，也没有 decode 任务可跑，那么scheduler 可以阻塞等待 tokenizer 发来新请求
若反之，那么scheduler 不能卡在收消息上，它会跳过 `self._process_one_msg()`，直接调用后面的`_schedule_next_batch`, `_forward`, 
以及 `_process_last_data`

总的来看 normal 的流程十分清晰: 
1. 接收用户的消息，对其进行某些处理，
2. 将整理消息组合成一个batch，
3. 进行 forward 推理，
4. 处理推理的结果

接下来，我们对这4个模块进行深入的解读

## Scheduler._process_one_msg
当收到了用户消息`msg`，会调用 `_process_one_msg(msg)`，代码如下:
```python
@dataclass
class UserMsg(BaseBackendMsg):
    uid: int
    input_ids: torch.Tensor  # CPU 1D int32 tensor
    sampling_params: SamplingParams

class Scheduler(SchedulerIOMixin):
    def _process_one_msg(self, msg: BaseBackendMsg) -> None:
        if isinstance(msg, BatchBackendMsg):
            for msg in msg.data:
                self._process_one_msg(msg)
        elif isinstance(msg, ExitMsg):
            raise KeyboardInterrupt
        elif isinstance(msg, UserMsg):
            logger.debug_rank0("Received user msg: %s", msg)
            input_len, max_seq_len = len(msg.input_ids), self.engine.max_seq_len
            max_output_len = max_seq_len - input_len
            if max_output_len <= 0:
                # 用户的输入太长了，无法推理，打印警告并退出

            if msg.sampling_params.max_tokens > max_output_len:
                msg.sampling_params.max_tokens = max_output_len


            self.prefill_manager.add_one_req(msg)
        elif isinstance(msg, AbortBackendMsg):
            # 打断执行
        else:
            logger.error(f"Unknown message type: {type(msg)}")
            raise NotImplementedError
```
上面代码中最核心的处理逻辑是 `UserMsg`的分支，这一分支调用了 `self.prefill_manager.add_one_req(msg)`


```python
class PrefillManager:
    # 省略其他的成员变量
    pending_list: List[PendingReq] = field(default_factory=list)

    def add_one_req(self, req: UserMsg) -> None:
        self.pending_list.append(PendingReq(req.uid, req.input_ids, req.sampling_params))
```
`self.prefill_manager.add_one_req(msg)`逻辑非常简单，只是利用 UserMsg 来构造 PendingReq，并将其加入到等待列表中

对比一下 `PendingReq` 与 `UserMsg` 的声明，前者最大的不同是比后者多了个 `chunked_req` 的字段，这个字段和大名鼎鼎的 `Chunked Prefill` 有关，后文会详细介绍`Chunked Prefill`。

```python
@dataclass
class PendingReq:
    uid: int
    input_ids: torch.Tensor
    sampling_params: SamplingParams
    chunked_req: ChunkedReq | None = None

    @property
    def input_len(self) -> int:
        return len(self.input_ids)

    @property
    def output_len(self) -> int:
        return self.sampling_params.max_tokens
```

## Scheduler._schedule_next_batch

在将消息加入到`pending_list`后，，`normal_loop` 会调用 `_schedule_next_batch()` 来生成一批新的输入，它的代码如下:
```python
    def _schedule_next_batch(self) -> ForwardInput | None:
        # TODO: support other policies: e.g. DECODE first
        batch = (
            self.prefill_manager.schedule_next_batch(self.prefill_budget)
            or self.decode_manager.schedule_next_batch()
        )
        return self._prepare_batch(batch) if batch else None
```

这个函数先通过 `prefill_manager` 或者 `decode_manager` 进行调度，然后对调度结果执行 `_perpare_batch`

由于 or 运算符的短路属性，当`prefill_manager.schedule_next_batch` 返回一个非 None 的 batch之后，
`decode_manager.schedule_next_batch`就不会执行了。因此，上面这段代码，如果 `prefill_manager.schedule_next_batch` 一直有结果，就会导致 `decode_manager.schedule_next_batch` 迟迟无法执行。

我们暂且不看 `decode_manger` 的调度逻辑，聚焦于`prefill_manager` 的调度逻辑


### PrefillManager.schedule_next_batch
直接上代码，下面的 ChunkedReq 和 ChunkedPrefill 有关，我们先注释掉它，
不看和它相关的逻辑，聚焦于一般情况下的调度
```python
class PrefillManager:
    cache_manager: CacheManager
    table_manager: TableManager
    # decode_manager 它在 prefill Manager 中的唯一作用就是提供inflight_tokens,
    # inflight_tokens = 所有 decode 请求剩余可生成 token 数 + 每个请求因 page 粒度带来的保守预留
    decode_manager: DecodeManager 
    pending_list: List[PendingReq] = field(default_factory=list)

    def schedule_next_batch(self, prefill_budget: int) -> Batch | None:
        if len(self.pending_list) == 0:
            return None
        
        # inflight_tokens的主要用于 prefill 调度时做容量判断：接纳新请求前，先把已有 decode 
        # 请求未来可能还要用的 KV cache 空间扣掉，避免新 prefill 请求把 cache 占满
        adder = PrefillAdder(
            token_budget=prefill_budget,
            reserved_size=self.decode_manager.inflight_tokens, 
            cache_manager=self.cache_manager,
            table_manager=self.table_manager,
        )
        reqs: List[Req] = []
        # chunked_list: List[PendingReq] = []
        for pending_req in self.pending_list:
            if req := adder.try_add_one(pending_req):
                pending_req.chunked_req = None
                # ChunkedReq 的特判逻辑，先不看了
                # if isinstance(req, ChunkedReq):
                #     pending_req.chunked_req = req
                #     chunked_list.append(pending_req)
                reqs.append(req)
            else:
                break  # We cannot add more requests

        if len(reqs) == 0:
            return None

        # self.pending_list = chunked_list + self.pending_list[len(reqs) :] # 假设不发生 Chunk,所以不看它
        self.pending_list = self.pending_list[len(reqs) :]
        return Batch(reqs=reqs, phase="prefill")
```

这个函数中最重要的逻辑是，新建一个`PefillAdder`的实例 `adder`, 然后从 `pending_list` 中逐个读取 
`pending_req`， 调用 `adder.try_add_one(pending_req)`，这个函数会返回一个 `Req` 类型的对象 
如果`adder.try_add_one` 返回 None，则停止遍历，并把已经生成的 reqs 构造成一个 `Batch` 返回。

```python
@dataclass
class Batch:
    reqs: List[Req]
    phase: Literal["prefill", "decode"]
    # these fields should be set by scheduler
    input_ids: torch.Tensor = field(init=False)
    positions: torch.Tensor = field(init=False)
    out_loc: torch.Tensor = field(init=False)
    padded_reqs: List[Req] = field(init=False)
    # this field should be set by attention backend
    attn_metadata: BaseAttnMetadata = field(init=False)
```

需要注意的一点是， Batch 除了 `reqs` 和 `phase` 两个成员外，还有另外五个重要成员，此时没被初始化呢。

显然 `adder.try_add_one` 是我们需要重点关注的函数，不过它内部用到了 `TableManager` 和 `CacheManager`，
我们需要先了解一下这两个类。


#### TableManager 剖析
`TableManager` 是**用来管理整个系统中处于推理状态的Req的**, 它控制处于推理状态的Req的数量，并且记录了这些
Req的每个token的值，以及每个token对应的KVCache的位置。

`TableManager`十分简洁，它的三个函数无需过多解释，重点是搞清楚 page_table 和 token_pool 这两个成员变量的含义
```python
class TableManager:
    def __init__(self, max_running_reqs: int, page_table: torch.Tensor) -> None:
        self._max_running_reqs = max_running_reqs # 默认是 256
        self._free_slots = list(range(max_running_reqs))
        self.page_table = page_table
        # NOTE: dummy request also use this pool to get the input ids, so we need to
        # make sure the token pool is initialized with valid values (token_id = 0).
        self.token_pool = torch.zeros_like(page_table, dtype=torch.int32)

    @property
    def available_size(self) -> int:
        return len(self._free_slots)

    def allocate(self) -> int:
        return self._free_slots.pop()

    def free(self, slot: int) -> None:
        self._free_slots.append(slot)

```

每个想要进入推理状态的Req都必须通过调用`table_manager.allocate`来分配一个`table_idx`,
这个`table_idx`就是 `page_table` 和 `token_pool` 的行号，每一行对应于一个Req

`page`是缓存系统中常见的概念，一般情况下，`page`是执行缓存驱逐的最小单位。在mini-sglang中，
**一个`page`内会存储一个到多个token的KVCache**，我们把每个token的KVCache称为`slot`,一个
`page`具体可以存储多少个`slot`，主要由`page size`来决定，在我们的代码解析过程中，不妨
假设 `page size = 4`

`page_table`是一个形状为(max_running_req + 1, aligned_max_seq_len)的Tensor, 
直观来看，page_table的样子如下
```
              seq pos:   0    1    2    3    4    5 ...
  table_idx 0          [  0,   1,   2,   3,  16,  17 ...]
  table_idx 1          [ 32,  33,  34,  35,  48,  49 ...]
  table_idx 2          [ ...                         ...]
  ...
  dummy row            [num_tokens, num_tokens, ...]
```
其实"page_table"这个名字起的不太恰当，因为它的值是 KV cache 里的一个`slot`的id，而不是一个page的id。换句话说，
`page_table[table_idx, pos]` 指向KVCache的某个 token slot。

至于 token_pool, 它的形状和 page_table 一样，不同点是，它的值是 token_id

#### RadixPrefixCache剖析
我们前面已经了解了`TableManager`，接下来要学习`CacheManager`，不过在学习它之前，我们需要了解一个重要的类，即
`RadixPrefixCache`。这个类是`CacheManager`的核心成员，有了它，Sglang才可以高效利用前缀信息。

##### RadixPrefixCache 的构造函数
关于 `RadixPrefixCache`的构造函数如下:
```python

```
在这个构造函数中，难以让人理解的有两个成员变量: `key_fn` 和 `root_node`

###### key_fn 是啥?
`self.key_fn`存储了`_get_key_fn`返回结果，后者的代码如下:
```python
KEY_FN: TypeAlias = Callable[[torch.Tensor], Any]

def _get_key_fn(page_size: int) -> KEY_FN:
    if page_size == 1:
        return lambda x: x[0].item()
    return lambda x: tuple(x[:page_size].tolist())
```
`_get_key_fn`返回的是一个匿名函数，这个匿名函数的输入是一个Tensor (这个Tenosr实际上是`Req.input_ids`)
匿名函数会返回`input_ids`的前`page_size`个Token组成的元祖，这个元祖将被 `RadixTreeNode`用来索引自己的
子节点

###### RadixTreeNode 是啥?
顾名思义，这个类是`RadixTree`的一个节点，网上有很多介绍基树的文章，这里就不多讲了
`RadixTreeNode`的源码才 70 多行，大部分函数非常简单，相信撕过树类算法题的读者可以秒懂
唯一值得拎出来分析的就是 `get_match_len`这个函数函数

```python
class RadixTreeNode:
    counter: int = 0

    def __init__(self, key_fn: KEY_FN, tic: int | None = None) -> None:
        self.key_fn = key_fn
        self.children: Dict[Any, RadixTreeNode] = {} # self.key_fn( childNode.key ) -> childNode
        self._parent: RadixTreeNode | None = None
        self.ref_count: int = 0
        self.uuid = RadixTreeNode.counter
        RadixTreeNode.counter += 1
        self.timestamp = tic or time.monotonic_ns()

        # these fields should be updated later
        self._key: torch.Tensor
        self._value: torch.Tensor # 存储的并不是KVCache的本体，而是KVCache的slot id
        self._length: int

    def set_key_value(self, key: torch.Tensor, value: torch.Tensor) -> None:
        assert len(key) == len(value)
        self._key = key
        self._value = value
        self._length = len(key)

    def set_parent(self, parent: RadixTreeNode) -> None:
        self._parent = parent
        parent.children[self.key_fn(self._key)] = self

    @property
    def length(self) -> int:
        return self._length

    @property
    def parent(self) -> RadixTreeNode:
        assert self._parent is not None
        return self._parent

    @property
    def value(self) -> torch.Tensor:
        return self._value

    def is_root(self) -> bool:
        return self._parent is None

    def is_leaf(self) -> bool:
        return len(self.children) == 0

    def get_match_len(self, input_ids: torch.Tensor) -> int:
        from minisgl.kernel import fast_compare_key

        # compare key and input_ids, find the first diff
        return fast_compare_key(self._key, input_ids)

    def split_at(self, pos: int) -> RadixTreeNode:
        assert 0 < pos < self.length
        parent = self.parent

        new_node = RadixTreeNode(self.key_fn, self.timestamp)
        new_node.set_key_value(self._key[:pos], self._value[:pos])
        new_node.set_parent(parent)
        new_node.ref_count = self.ref_count

        self.set_key_value(self._key[pos:], self._value[pos:])
        self.set_parent(new_node)

        return new_node

    # lt 意为 lower than, 它重载了 < 操作符，每个 node 按照创建时的时间戳进行大小比较
    def __lt__(self, other: RadixTreeNode) -> bool:
        return self.timestamp < other.timestamp
```

`get_match_len` 实际上是通过 cpp 函数 fast_compare_key 实现的，代码如下:

```python
def fast_compare_key(x: torch.Tensor, y: torch.Tensor) -> int:
    # compare 2 1-D int cpu tensors for equality
    return _load_radix_module().fast_compare_key(x, y)
```

```cpp
auto fast_compare_key(const tvm::ffi::TensorView a,
                      const tvm::ffi::TensorView b) -> size_t {
  const auto a_ptr = a.data_ptr();
  const auto b_ptr = b.data_ptr();
  const auto common_len = std::min(a.size(0), b.size(0));
  if (a.dtype().bits == 64) { // 64 位机器
    const auto a_ptr_64 = static_cast<const int64_t *>(a_ptr);
    const auto b_ptr_64 = static_cast<const int64_t *>(b_ptr);
    const auto diff_pos =
        std::mismatch(a_ptr_64, a_ptr_64 + common_len, b_ptr_64);
    return static_cast<size_t>(diff_pos.first - a_ptr_64);
  } else { // 32 位机器
    const auto a_ptr_32 = static_cast<const int32_t *>(a_ptr);
    const auto b_ptr_32 = static_cast<const int32_t *>(b_ptr);
    const auto diff_pos =
        std::mismatch(a_ptr_32, a_ptr_32 + common_len, b_ptr_32);
    return static_cast<size_t>(diff_pos.first - a_ptr_32);
  }
}
```

可见，`fast_compare_key`就是在**计算两个序列的相同前缀长度**
所以，`RadixTreeNode.get_match_len(input_ids)`是在计算`input_ids`与节点自己的`key`之间
相同前缀的长度

##### RadixPrefixCache.match_prefix
在对`key_fn`和`RadixTreeNode`有了全面认识之后，我们来读 `match_prefix` 的源码,
`match_prefix`实际上只是对 `_tree_walk` 的薄封装

```python
    def match_prefix(self, input_ids: torch.Tensor) -> MatchResult:
        node, prefix_len = self._tree_walk(input_ids)
        return MatchResult(RadixCacheHandle(prefix_len, node))

    def _tree_walk(self, input_ids: torch.Tensor) -> Tuple[RadixTreeNode, int]:
        prefix_len = 0
        indice_len = len(input_ids)
        node = self.root_node
        tic = time.monotonic_ns()

        while prefix_len < indice_len:
            # 疑问， input_ids[ prefix_len: ] 的长度可能不足 page_size 吗？
            child_node = node.children.get(self.key_fn(input_ids[prefix_len:]))
            if child_node is None:
                return node, prefix_len
            node = child_node  # walk to child node

            # NOTE: at least 1 page is matched, so match_len >= page_size
            match_len = node.get_match_len(input_ids[prefix_len:])
            match_len = align_down(match_len, self.page_size)
            prefix_len += match_len

            # need to split the node if not fully matched
            if match_len != node.length:
                node = node.split_at(match_len)
                node.timestamp = tic
                return node, prefix_len

            # update timestamp for accessed node
            node.timestamp = tic

        return node, prefix_len
```

`self._tree_walk(input_ids)` 就是从基树缓存的根节点出发，将`input_ids`与基树缓存节点尽可能多地匹配，
返回的Node是末尾前缀的Node(这个Node通过追寻parent变量即可查找到所有前缀 token 的Node)，换句话说，
利用`_tree_walk(input_ids)`的返回值可以找到缓存中与`input_ids`匹配的最长前缀的`KVCache`

得到`_tree_walk`的结果后，match_prefix 又用这个结果构造了一个 `RadixCacheHandle`，这个类只有一个方法，那就是
`get_matched_indices`，这个函数通过从叶子节点一路追踪到根节点，来构造KVCache的 slot id 序列
```python
class BaseCacheHandle(ABC):
    cached_len: int

    @abstractmethod
    def get_matched_indices(self) -> torch.Tensor: ...

class RadixCacheHandle(BaseCacheHandle):
    node: RadixTreeNode

    def get_matched_indices(self) -> torch.Tensor:
        node = self.node
        value_list: List[torch.Tensor] = []
        while not node.is_root():
            value_list.append(node.value)
            node = node.parent
        value_list.reverse()
        return torch.cat(value_list)
```

match_prefix进一步用RadixCacheHandle构造了MatchResult，而MathcResult也只是一层薄包装罢了
```python
class MatchResult(NamedTuple):
    cuda_handle: BaseCacheHandle
```

##### RadixPrefixCache.lock_handle
前面我们介绍了 `match_prefix`, 通过`match_prefix`查询得到一个 `RadixCacheHandle` 之后，
需要先调用 `lock_handle`来确保相关的`RadixTreeNode`不被驱逐，才能继续安全地使用它们来查找`KVCache`
`lock_handle`的实现如下，从子节点向父节点的方向遍历，增加或减少每个 node 的 ref_count，来实现上锁或解锁
只有当`ref_count`为0时，`RadixTreeNode`才能被驱逐出缓存

```python
    def lock_handle(self, handle: BaseCacheHandle, unlock: bool = False) -> None:
        assert isinstance(handle, RadixCacheHandle)
        node = handle.node
        if unlock:
            while not node.is_root():
                node.ref_count -= 1
                assert node.ref_count >= 0
                if node.ref_count == 0:
                    self.evictable_size += node.length
                    self.protected_size -= node.length
                node = node.parent
        else:
            while not node.is_root():
                if node.ref_count == 0:
                    self.evictable_size -= node.length
                    self.protected_size += node.length
                node.ref_count += 1
                node = node.parent
```

##### RadixPrefixCache.evict
当内存不够用时，就该调用 `evict` 进行缓存驱逐了，代码如下(删除了部分边界条件的分支）
```python
    # 目标是驱逐至少size个Page的KVCache，并将产生的空闲 page 返回
    def evict(self, size: int) -> torch.Tensor:
        if size == 0:
            return self.empty_tensor
        # 收集 基树 中所有引用计数为0的叶子节点，引用计数为0意味着可驱逐 
        leave_nodes = self._collect_leave_nodes_for_evict() 

        # leave_nodes 是列表，heapq是一个python标准库，heapq.heapify( list )的作用是 将list重排成一个小根堆
        # 前面提到过，RadixTreeNode 重载了 __lt__ 函数，RadixTreeNode之间按照创建时间戳的早晚决定大小，
        # 一个RadixTreeNode 创建得越早，它就越小
        heapq.heapify(leave_nodes) 
        evicted_indices: List[torch.Tensor] = []
        evicted_size = 0

        # 按照创建时间戳从早到晚来驱逐引用计数为0的RadixTreeNode
        while evicted_size < size:
            node = heapq.heappop(leave_nodes)
            assert node.ref_count == 0 and node.is_leaf() and not node.is_root()
            evicted_size += node.length
            evicted_indices.append(node.value) # node.value 是一个序列，序列的每一个元素都对应于一个KVCache Pool 的物理槽位
            self.evictable_size -= node.length
            parent = node.parent
            del parent.children[self.key_fn(node._key)]
            # 不用担心下面的代码会删除 root ，因为 root 的 ref_count 永不为0
            if parent.is_leaf() and parent.ref_count == 0:
                heapq.heappush(leave_nodes, parent)

        return torch.cat(evicted_indices)
```
从上可以看出，所谓的 evict，其实就是按照创建时间从老到新的顺序，从叶子节点开始，依次删除所有的引用计数为0的节点。
然后将驱逐后留下的KVcache Pool的slot id列表 (也就是 evicted_indices )返回

##### RadixPrefixCache.inser_prefix
前面讲了从 RadixPrefixCache中查询和驱逐 KVCache的过程，接下来介绍的是往 RadixPrefixCache中插入KVCache的过程。
代码如下：

前四行代码进行了将KVCache真正插入 RadixTree 中的关键操作，它从 req.input_ids 中取出了 req.cache_len 长度的 tokenid 及对应的KVCache Slot
将它们作为参数，执行了 prefix_cache.insert_prefix:

```python
class RadixPrefixCache(BasePrefixCache):
    def insert_prefix(self, 
                    input_ids: torch.Tensor, # 可以将它想象成完成Prefill 阶段后的 token id 序列
                    indices: torch.Tensor # 这是 input_ids 对应的 KVCache 的 slot_id 序列，满足len(indices) == len(input_ids)
        ) -> InsertResult:
        # 关键操作，讲 input_ids 和 incides 的长度向下截取到 page_size 的整数倍
        insert_len = align_down(len(input_ids), self.page_size)
        input_ids, indices = input_ids[:insert_len], indices[:insert_len]
        # 查找当前 RadixPrefixCache 中与 input_ids 匹配的最长前缀
        node, prefix_len = self._tree_walk(input_ids)
        if prefix_len != insert_len:  # NOTE: prefix_len < insert_len
            new_node = RadixTreeNode(self.key_fn)
            new_node.set_key_value(input_ids[prefix_len:], indices[prefix_len:].clone())
            new_node.set_parent(node)
            self.evictable_size += new_node.length
            node = new_node
        return InsertResult(prefix_len, RadixCacheHandle(insert_len, node))
```
`insert_prefix` 中首先执行了一个让人有些诧异的动作，它对 `input_ids` 和 `indices`进行了截取，使得它的长度是`page_size`的整数倍

这是因为 mini-sglang 的 KV Cache 是以整页为最小分配、共享和回收单位的。RadixPrefixCache 只能安全缓存完整页，不能缓存最后那个未填满的页

为了更好地理解这一点，我们举个反例，假设 page_size = 4，一个前缀有 6 个 token,
并假设允许往 RadixPrefixTree中存储未填满的页，那么此时
```
page 0: token 0 1 2 3    完整
page 1: token 4 5 _ _    不完整
```
如果此时线程A 想在 page 1 的位置 2 写 token X, 线程 B 想在 page 1 的位置 2 写 token Y, 二者就会写同一个物理 KV 页，发生覆盖。

在进行截取后，将截取后的 input_ids 的KVcache slot全都存到了基数缓存中，这部分代码用到的关键函数前面有梳理过，就不细讲了

最后，返回了一个 InsertResult，需要注意的是，InsertResult 的第一个字段是**执行此次插入之前，缓存中与req.input_ids的最大前缀长度**，
后面我们将理解为什么需要用到这个字段

通过读这段代码，我们可以得知一个关键信息: **基树缓存中的KVCache是以页为最小单位的，每个基树的节点保留的 token 和 KVCache slot 数，都是
page_size 的整数倍。并且每 page_size 个相邻的token，都被分配到了同一个page上**

#### CacheManager 剖析
前面，我们介绍了 `RadixPrefixCache` 的查询缓存(`match_prefix`)和驱逐缓存(`evict`)的过程，接下来就可以来介绍
`CacheManager`了。
顾名思义，`CacheManager`负责管理KVCache，它的结构则比`TableManager`的要复杂得多，我们需要拆分成好几小节来介绍。

##### CacheManager 的构造函数
CacheManager 的构造函数如下，其中`page_table`这一变量的含义与`TableManager`中的`page_table`完全一样
`create_prefix_cache` 则会根据 type 的值创建一个 `RadixPrefixCache` 或者 `NaivePrefixCache` 实例
后者的实现非常简单，我们可以忽略它。

构造函数中仍有一点需要注意，那就是`CacheManager.free_slots`实际上对应的是每个KVCache页的第一个slot，所以实际上可以
将`free_slots`理解为`free_pages`

```python
class CacheManager:
    def __init__(self, num_pages: int, page_size: int, page_table: torch.Tensor, type: str):
        # The `_free_slots` follows a page-aligned manner. For example, if page_size = 2,
        # the `_free_slots` may look like [0, 2, 4, 6, ...], and each slot represents a page.
        device = page_table.device
        self.free_slots = torch.arange(num_pages, dtype=torch.int32, device=device) * page_size

        # type 是指 cache_type，取值是 radix 和 native
        self.prefix_cache = create_prefix_cache(device=device, type=type)
        self.device = device
        self.num_pages = num_pages
        self.page_table = page_table
        self.page_size = page_size
```

##### CacheManager.match_req
`CacheManager`的一个重要函数就是 `match_req`，代码如下：
```python
class CacheManager:
```

它实际上调用了底层的`RadixPrefixCache`的 `match_req`，前面我们已经分析过了，它会返回当前基树中与

##### CacheManager.allocate_paged
这是一个复杂的函数，代码如下，如果你看不懂的话，可以先跳过，等后文用到 `allocated_paged`时，再来读它:
```python
    def allocate_paged(self, reqs: List[Req]) -> None:
        needed_pages = 0
        allocation_info: List[Tuple[int, int, int]] = []
        for req in reqs:
            # div_ceil : 对两数相除的结果向上取整
            # req.cached_len 是目前基树缓存与req.input_ids的最长共同前缀数
            first_page = div_ceil(req.cached_len, self.page_size) 
            # 在 Req.__post_init里，device_len 被赋值为 len(req.input_ids)
            last_page = div_ceil(req.device_len, self.page_size)  
            if last_page > first_page: # 需要分配新的 Page 来承接推理过程中产生的 KVCache
                # 记录这些待分配的 page 信息，后面统一分配
                needed_pages += last_page - first_page
                allocation_info.append((req.table_idx, first_page, last_page))
        if needed_pages > 0:
            allocated = self._page_to_token(self._allocate(needed_pages))
            _write_page_table(self.page_table, allocated, allocation_info, self.page_size)
```
它的参数是一个推理批次里的所有 Req，这个函数遍历了每一个 Req，计算它们本次 Prefill 产生的新KVCache所需要的页数，
然后为这些即将产生的KVCache提前分配所需的页。
页分配的过程涉及 `_allocate`, `_page_to_token`, `_write_page_table` 三个函数，下面分别进行讲解

###### CacheManager._allocate
代码如下，简短且清晰，主要流程是`self.free_slots` 取出指定数目的 `page_id`，返回给调用者(回忆一下CacheManager的构造函数，你还记得
它的 `free_slots` 长什么样子吗？)
如果当前的 `free_slots` 不够用，则调用 `self.prefix_cache.evict` 驱逐出足量的 KVCache(这个函数的内容前面已经讲过了)
```python
    def _allocate(self, needed_pages: int) -> torch.Tensor:
        if needed_pages > (free_pages := len(self.free_slots)):
            evicted = self.prefix_cache.evict((needed_pages - free_pages) * self.page_size)
            # 注意， evicted 内是 slot_id 序列，所以需要下面这样代码来将其转化成 page_id
            self.free_slots = torch.cat([self.free_slots, evicted[:: self.page_size]])
            assert len(self.free_slots) >= needed_pages, "Eviction did not free enough space."
        allocated = self.free_slots[:needed_pages]
        self.free_slots = self.free_slots[needed_pages:]
        return allocated
```

###### CacheManager._page_to_token
`_page_to_token`的输入是`_allocate`返回的 page_id 列表, `_page_to_token`的作用是将每个 page 的起始 slot
展开成这个 page 内所有 token 对应的 slot_id

```python
    def _page_to_token(self, pages: torch.Tensor) -> torch.Tensor:
        if self.page_size == 1:
            return pages
        # [X * page_size] -> [X * page_size, ..., X * page_size + page_size - 1]
        offsets = torch.arange(self.page_size, device=self.device, dtype=torch.int32)
        return (pages.unsqueeze(1) + offsets).flatten()
```
假设 self.page_size = 4, pages = tensor([0, 4, 12])
那么这个函数会返回 [0, 1, 2, 3, 4, 5, 6, 7, 12, 13, 14, 15]

###### CacheManager._write_page_table
`_write_page_table`这个函数的作用是把已经分配的KV Cache slot 写入到 `page_table` 中
具体应该写到`page_table`的哪个位置？写入位置是通过 allocation_info 计算得到的，感兴趣的
读者可以仔细研读它的代码，下面的注释有助于你顺利读懂
```python
def _write_page_table(
    page_table: torch.Tensor, # 形状是(config.max_running_req + 1, aligned_max_seq_len)，存储 token 对应的 KVCache slot id
    allocated: torch.Tensor, # 是 _page_to_token 的返回值，一个一维的Tensor，每个元素都是一个 KVCache 的 Slot id
    allocation_info: List[Tuple[int, int, int]], # 每个元素都是一个(table_id, first_page, last_page)，三元组，意思是需要给 table_id 对应的 req 的 [first_page, last_page) 分配内存
    page_size: int,
) -> None:
    needed_tokens = len(allocated)
    # table_idx_host是 page_table 的行索引，table_idx_host[i]: 第i个 slot 应该被分配给哪个 Req
    table_idx_host = torch.empty(needed_tokens, dtype=torch.int64, pin_memory=True) 
    # positions_host是 page_table 的列索引，positions_host[i]: 第i个 slot 应该被分配给这个Req的哪个pos上？
    positions_host = torch.empty(needed_tokens, dtype=torch.int64, pin_memory=True) 

    offset = 0
    for table_idx, first_page, last_page in allocation_info:
        first_pos, last_pos = first_page * page_size, last_page * page_size
        length = last_pos - first_pos
        table_idx_host[offset : offset + length].fill_(table_idx)
        torch.arange(first_pos, last_pos, out=positions_host[offset : offset + length])
        offset += length
    assert offset == needed_tokens, "Mismatch in allocated tokens and filled tokens."
    table_idxs = table_idx_host.to(page_table.device, non_blocking=True)
    offsets = positions_host.to(page_table.device, non_blocking=True)
    page_table[table_idxs, offsets] = allocated
```

###### 总结
通过上面3个函数的解析，我们已经清晰地了解了allocated_paged的过程。
1. 通过比较 req.cached_len, req.device_len的长度来决定是否要新分配page
2. 若需要新分配，则从 free_slot 中分配page，如果 free_slot 不够用，则按照从老到新的顺序驱逐`RadixPrefixCache`中的叶节点
3. 最后，在`page_table`中记录被新分配的 KVCache slot id



##### CacheManger.cache_req
`CacheManger.cache_req`一般在完成一轮推理后调用，它是用来将新产生的KVCache slot id存到`RedixPrefixCache`里的:
```python
    def cache_req(self, req: Req, *, finished: bool) -> None:
        # ==================================== valid cache region ====================================
        # [0, req.cached_len)                       This part is valid for attention kernel read/write.
        # [0, old_handle.cached_len)                This part is in the prefix cache before prefill.
        # [old_handle.cached_len, req.cached_len)   This part is allocated by cache manager for this request.
        # ================================== allocated cache region ==================================
        # [old_handle.cached_len, cached_len)       This part was not in the prefix cache when prefill,
        #                                           but later cached by other requests.
        #                                           We must free them to avoid memory leak.
        # [cached_len, new_handle.cached_len)       This part is newly inserted into the prefix cache.
        # [new_handle.cached_len, req.cached_len)   This part is tailing part that can not inserted into the prefix cache.
        #                                           We should free it if the request has finished.
        insert_ids = req.input_ids[: req.cached_len]
        page_indices = self.page_table[req.table_idx, : req.cached_len]
        old_handle = req.cache_handle
        cached_len, new_handle = self.prefix_cache.insert_prefix(insert_ids, page_indices)
        # unlock until all operations on handle is done
        self.unlock(old_handle)
        # this part is already in the prefix cache, free it
        self._free(page_indices[old_handle.cached_len : cached_len])
        if finished:  # this tail part should be freed
            self._free(page_indices[new_handle.cached_len :])
        else:  # keep the tail part, update the handle
            req.cache_handle = new_handle
            self.lock(new_handle)
```
它的req参数是完成了一轮推理后的req，此时`req.cache_len`包括了此轮推理新产生的KVCache，所以在`CacheManger.cache_req`的一开始，通过
`req.cache_len`截断了 `insert_ids`和`page_table[req.table_idx]`。

在完成截断之后，`CacheManger.cache_req` 将截断得到的 `input_ids` 及对应的 KVCache slot_id 的列表作为参数，通过调用`prefix_cache.insert_prefix`
来尝试将新的KVCache插入基树缓存中。
在完成插入之后，随后又对 [old_handle.cached_len: cached_len) 区间内的 slot_id进行了释放。
这个操作乍一看很奇怪,为什么要释放这段KVCache呢？
这是因为它其实是冗余的了
old_handle.cached_len 的含义是 **在对这个Req进行推理前，RadixPrefixCache中与Req.input_ids的最长共同前缀**
cache_len 的含义在 `RadixPrefixCache.inser_prefix` 中强调过了，是**在对这个Req进行推理后，在插入推理产生的KVCache前，RadixPrefixCache中与Req.input_ids的最长共同前缀**

因此， [old_handle.cached_len: cached_len) 这段区间内的KVCache，已经被别的Req插入`RadixPrefixCache`了，所以当前Req的这段KVCache就
冗余了，因此释放掉。

最后的5行代码非常简单，如果推理已经完成，则将多余分配的 KVCache slot 回收。如果没有完成，则更新一下 req 中的 cache_handle

###### Engine._sync_get_memroy
该函数内部涉及一点分布式通信，这里就不贴出代码细讲了，直接介绍它的运行逻辑。

这个函数通过网络通信，计算出整个通信组中的Rank之间的最小显存和最大显存，如果最大显存和最小显存之差大于
2GB，则直接报错（这是因为 mini-sglang 的每个Rank都按照最大显存来计算可用page数，最小显存的Rank上计算
的可用page数不能超过实际可用page数太多，否则可能导致内存分配失败）

`_sync_get_memory()`返回的是 (最小显存，最大显存) 的二元组。 `init_free_memory = self._sync_get_memory()[1]`
实际上就是把每个Rank上的可用显存统一设置为最大显存

###### Engine._determine_num_pages
```python
    def _determine_num_pages(self, old_free_memory: int, config: EngineConfig) -> int:
        new_free_memory = self._sync_get_memory()[1]
        cache_per_page = (
            2  # key + value
            * config.model_config.head_dim # 做多头注意力时，每个 token 的 embedding 维数
            * div_even(config.model_config.num_kv_heads, config.tp_info.size, allow_replicate=True) # 每个 Rank 应该承担的 head 数
            * config.page_size # page_size，每个page应该存储 page_size 个KVcache slot id
            * self.dtype.itemsize # dtype 的字节数，例如 float16 应该返回2， float32 应该返回4
            * config.model_config.num_layers # 模型的层数，每层都意味着一次 attention
        )

        num_pages = config.num_page_override # 预先配置的page数
        if num_pages is None:
            model_memory = old_free_memory - new_free_memory
            available_memory = int(config.memory_ratio * old_free_memory) - model_memory
            num_pages = available_memory // cache_per_page

        assert num_pages > 1, "Not enough memory for KV cache, try reducing --num-pages"
        num_tokens = num_pages * config.page_size
        real_kv_size = num_pages * cache_per_page
        logger.info(f"Allocating {num_tokens} tokens for KV cache, K + V = {mem_GB(real_kv_size)}")
        return num_pages
```
这个函数打一开始就调用 `new_free_memory = self._sync_get_memory()[1]` 重新计算了一下可用内存，这是因为 `old_free_memory`
是模型加载前的，而 `new_free_memory`才是模型加载后真正用的内存。

接下来的 `cache_per_page` 计算的是一个KVCache的page所占用的字节数。我把每个乘项的含义都写在了注释里，如果读者感到理解困难的话，
不妨去手撕一个Transformer，等能够轻松理解 `cache_per_page`的计算流程后，再接着往下看。

在这里，我们稍微仔细地看一下`div_even(config.model_config.num_kv_heads, config.tp_info.size, allow_replicate=True)`
`config.model_config.num_kv_heads`是指当前的模型做多头注意力的头数，`config.tp_info.size`则是指通信组中的节点数量
`allow_replicate=True` 意味着，当 `进程中组中的节点数量 > 大模型推理的多头注意力头数`时，每个节点至少会承担一个头的计算任务.

```python
def div_even(a: int, b: int, allow_replicate: bool = False) -> int:
    """Divides two integers. If allow_replicate=True, allows b > a when b % a == 0, returning 1."""
    if allow_replicate and b > a:
        assert b % a == 0, f"{b = } must be divisible by {a = } for KV head replication"
        return 1
    assert a % b == 0, f"{a = } must be divisible by {b = }"
    return a // b
```

完成`cache_per_page`的计算后，如果用户没有预先配置page数的话，就可以基于显存来计算page数了:
```python
        model_memory = old_free_memory - new_free_memory
        available_memory = int(config.memory_ratio * old_free_memory) - model_memory
        num_pages = available_memory // cache_per_page
```
上面的 available_memory 的计算可能有点奇怪，不过将它的计算整理一下就得到了:
`avalilable_memory = 加载模型后的空闲显存 - (1 - memory_ratio) × 加载模型前的空闲显存`
默认 memory_ratio 是 0.9，也就是基于加载模型前的空闲显存，预留约 10% 不给模型和 KV cache 使用。
比如加载前空闲 40 GiB、加载后空闲 30 GiB，就给 KV cache 约 30 - 4 = 26GiB 的预算

为什么要预留?
因为 KV cache 不是初始化和推理过程中最后一笔显存开销。在它分配之后，这个项目还会创建页表、attention/MoE 后端、采样器，并捕获 CUDA Graph
运行时也可能需要临时Tensor和算子工作区。若把测到的空闲显存全部分给 KV cache，后续任意一次分配都可能 OOM

##### MHAKVCache 详解

```python
class MHAKVCache(BaseKVCachePool):
    """
    Base class for key-value caches.
    This class defines the interface for key-value caches used in LLMs.
    """

    def __init__(
        self,
        num_kv_heads: int,
        num_layers: int,
        head_dim: int,
        num_pages: int,
        page_size: int,
        dtype: torch.dtype,
        device: torch.device,
    ) -> None:
        tp_info = get_tp_info()
        local_kv_heads = div_even(num_kv_heads, tp_info.size, allow_replicate=True)
        self._kv_buffer = torch.empty(
            (2, num_layers, num_pages, page_size, local_kv_heads, head_dim),
            device=device,
            dtype=dtype,
        )
        self._num_layers = num_layers
        self._k_buffer = self._kv_buffer[0]
        self._v_buffer = self._kv_buffer[1]
        self._device = device
        self._storage_shape = (num_pages * page_size, local_kv_heads, head_dim)

    def k_cache(self, index: int) -> torch.Tensor:
        return self._k_buffer[index]

    def v_cache(self, index: int) -> torch.Tensor:
        return self._v_buffer[index]

    def store_kv(
        self, k: torch.Tensor, v: torch.Tensor, out_loc: torch.Tensor, layer_id: int
    ) -> None:
        from minisgl.kernel import store_cache

        store_cache(
            k_cache=self._k_buffer[layer_id].view(self._storage_shape),
            v_cache=self._v_buffer[layer_id].view(self._storage_shape),
            indices=out_loc,
            k=k,
            v=v,
        )

    @property
    def device(self) -> torch.device:
        return self._device

    @property
    def dtype(self) -> torch.dtype:
        return self._kv_buffer.dtype

    @property
    def num_layers(self) -> int:
```

可见，`store_kv` 只是封装了 `store_cache`，而后者的实现如下，关键的信息已经以注释的形式给出，
```python
def store_cache(
    k_cache: torch.Tensor, # shape: (num_pages * page_size, local_kv_heads, head_dim)
    v_cache: torch.Tensor, # shape: (num_pages * page_size, local_kv_heads, head_dim)
    indices: torch.Tensor, # shape: (N, )
    k: torch.Tensor, # shape: [N, local_kv_heads * head_dim]
    v: torch.Tensor, # shape: [N, local_kv_heads * head_dim]
) -> None:
    num_tokens = k_cache.shape[0] # num_tokens = num_pages * page_size
    k_cache = k_cache.view(num_tokens, -1) # shape: (num_pages * page_size, local_kv_heads * head_dim)
    v_cache = v_cache.view(num_tokens, -1) # shape: (num_pages * page_size, local_kv_heads * head_dim)
    element_size = k_cache.shape[1] * k_cache.element_size() # 计算 k_cache 一行数据所占的字节数 
    module = _jit_store_module(element_size) # 用 element_size作为模板参数，实例化了一个cpp模板类，模板类的名字是 StoreKernel
    module.launch(k_cache, v_cache, indices, k, v) # 运行 StoreKernel<...>.run
```
最后两行代码内部涉及了一些繁琐的细节，我们就不详细介绍了，读者只需记住，最后两行代码实例化了一个叫做`StoreKernel`的cpp模板，并且调用了它的
`run`函数即可。接下来，我们就来一窥这个 `StoreKenel` 的真容:

```cpp
template <std::size_t element_size, // depends on data type and embedding dim
          std::size_t num_threads = 128,   // number of threads per block
          std::size_t max_concurrency = 1, // max blocks per SM
          bool use_pdl = false>
struct StoreKernel {
  static void run(const tvm::ffi::TensorView k_cache,
                  const tvm::ffi::TensorView v_cache,
                  const tvm::ffi::TensorView indices,
                  const tvm::ffi::TensorView k, 
                  const tvm::ffi::TensorView v) {
    // 一些很trick的代码，检查了5个参数是否合法，
    // 并且计算了 kv_cache_stride, kv_input_stride, length。省略这些trick的代码

    const auto params = StoreKernelParams{
        .k_cache = k_cache.data_ptr(),
        .v_cache = v_cache.data_ptr(),
        .indices = indices.data_ptr(),
        .k = k.data_ptr(),
        .v = v.data_ptr(),
        /** 
            stride 是在张量中一个很重要的概念，意为沿某个维度前进一步，底层存储地址要跳过多少个元素
            例如，如果一个张量 A 的形状是 [2, 3, 4]，那么它的 strides = [24, 12, 1]
            A[i ,j ,k] 底层就通过 24 * i + 12 * j + 1 * k 来计算 A[i, j, k] 相对 A[0, 0, 0] 的偏移
        **/
        .kv_cache_stride = kv_cache_stride, // 是一个整数，这是因为 k_cache 是二维的，所以它的 strides = [N, 1]，可以简记为 N
        .kv_input_stride = kv_input_stride, // 同样也是一个整数
        .length = length, // length = len(indices)，即要写入的token数
    };

    // 代码设定一个 block 有 num_threads 个线程，而在GPU中，一个Wrap是32个线程
    constexpr auto kWarpPerBlock = num_threads / 32;
    static_assert(num_threads % 32 == 0);
    // 此项目中，每一个wrap负责将一个token的KVCache写入KVCache Pool 中，
    // 因此用 (length // kWarpPerBlock ) 来计算需要的block数
    const auto num_blocks = div_ceil(length, kWarpPerBlock);
    const auto kernel = use_int32
                            ? store_kv_cache<num_threads, max_concurrency, 
                                             use_pdl, element_size, int32_t>
                            : store_kv_cache<num_threads, max_concurrency,
                                             use_pdl, element_size, int64_t>;
    LaunchKernel(num_blocks, num_threads, device)
        .with_attr(use_pdl)(kernel, params);
  }
};
```
`StoreKernel::run`的开头有一大段的参数校验和计算的逻辑，为了不陷入繁琐的无关细节中，这里将它删去了。
剩余代码中，笔者也给出了关键注释。`StoreKernel::run` 这个函数调用了`store_kv_cache`这个cuda核函数，
注释中包含了一些帮助读者理解代码的信息
```cpp
template <std::size_t kNumThreads, std::size_t kMaxOccupancy, bool kUsePDL, 
            std::size_t kElementSize, std::integral T>
__global__ __launch_bounds__(kNumThreads, kMaxOccupancy) void 
    store_kv_cache(const __grid_constant__ StoreKernelParams params) {
    // __launch_bounds__(kNumThreads, kMaxOccupancy) 管理线程块大小和资源分配。
    // 第一个参数表示这个 kernel 启动时每个 block 最多有 kNumThreads 个线程；第二个参数表示希望每个 SM至少能同时驻留 kMaxOccupancy 个 block。
    // 编译器会据此考虑寄存器用量。

    // __grid_constant__ 管参数 params 本身。这里整个 StoreKernelParams 结构体按值传入 kernel；标记后，
    // 同一次 kernel 启动中的线程可以通过同一个只读地址访问它，编译器不必因取参数地址而给每个线程造一份局部副本
  using namespace device;

  constexpr auto kWarpPerBlock = static_cast<unsigned>(kNumThreads / kWarpThreads);
  static_assert(kNumThreads % kWarpThreads == 0);

    // k_cahce 和 v_cache 的 shape: [num_page * page_size, local_kv_heads * head_dim]
    // k, v的shape: [N, local_kv_heads * head_dim]
  const auto &[k_cache, v_cache, indices, k, v, kv_cache_stride, kv_input_stride, length] = params;
  const auto warp_id = (threadIdx.x / kWarpThreads) + blockIdx.x * kWarpPerBlock;

    // PDL 是 CUDA 的 Programmatic Dependent Launch，用于控制同一 stream 中有依赖关系的 kernel 何时可以开始执行
    // PDL::wait()，确保所依赖的前一个 kernel 已经完成；做完 K/V 拷贝后调用 PDL::launch()，通知后续依赖的 kernel 可以启动
    // kUsePDL为 true 时，kernel 内的 PDL::wait() 和 PDL::launch() 才会生成对应指令；为 false 时，这两处基本是空操作
  PDL::wait<kUsePDL>();

  // each warp handles one element
  if (warp_id < length) {
    // indices就是前面的out_loc, 如果indices[i]的值是j，表示k[i]应该写入到k_cache[j]
    const auto pos = static_cast<const T *>(indices)[warp_id]; 
    const auto dst_k = pointer::offset(k_cache, pos * kv_cache_stride);
    const auto src_k = pointer::offset(k, warp_id * kv_input_stride);
    // kElementSize是 element_size == dtype_size * local_kv_heads * head_dim，即 k或v的一行数据的字节数
    warp::copy<kElementSize>(dst_k, src_k); 
    const auto dst_v = pointer::offset(v_cache, pos * kv_cache_stride);
    const auto src_v = pointer::offset(v, warp_id * kv_input_stride);
    warp::copy<kElementSize>(dst_v, src_v);
  }

  PDL::launch<kUsePDL>();
}
```

可见`store_kv_cache`指定每个Wrap拷贝一个token的KVCache，拷贝的具体逻辑在`warp::copy`中，其中`kElementSize`是每个
Warp需要拷贝的字节数。`warp::copy`的逻辑如下，线程根据自己的`lane_id`来搬运自己负责的数据
```cpp

// kUnit的值是根据 kBytes 的大小决定的，kUnit的合法取值是 16, 8, 4 中的一个
template <std::size_t kBytes,
          std::size_t kUnit = details::resolve_unit_size(kBytes)>
__always_inline __device__ void copy(void *__restrict__ dst,
                                     const void *__restrict__ src) {
    // details::mem_package_t根据kUnit来决定 Package的具体大小
    // 如果kUnit为16，则Package是一个16字节大小的结构体，kUnit为8，则是8字节的结构体，kUnit为4时同理。。。
  using Package = details::mem_package_t<kBytes, kUnit>;
  // 一个Warp中一次循环所能搬运的字节数
  constexpr auto kBytesPerLoop = sizeof(Package) * kWarpThreads;
  // 需要进行kLoopCount此循环搬运
  constexpr auto kLoopCount = kBytes / kBytesPerLoop;
  static_assert(kBytes % kBytesPerLoop == 0,
                "kBytes must be multiple of 128 bytes");

  const auto dst_packed = static_cast<Package *>(dst);
  const auto src_packed = static_cast<const Package *>(src);
  // 注意看这里，一个Wrap中，不同线程的 lane_id 是不同的
  const auto lane_id = threadIdx.x % kWarpThreads;

#pragma unroll kLoopCount // 提示编译器循环展开
  for (std::size_t i = 0; i < kLoopCount; ++i) {
    const auto j = i * kWarpThreads + lane_id;
    dst_packed[j] = src_packed[j];
  }
}
```

总的来说，mini-sglang在将KVCache存入 KVCachePool中时，一个token对应的 k 和 v 是由32个线程来合作搬运的。
一个 k 有 local_head_num * head_dim 个元素


#### PrefillAdder.try_add_one 剖析
前面`PrefillManager.schedule_next_batch`中调用了`PrefillAdder.try_add_one`，为了方便读懂这个函数，
我们花了大量的时间去学习 `TableManger`, `RadixPrefixCache` 以及 `CacheManager`。我们还顺便剖析了
`Engine`中的`MHAKVCache`的细节过程。

有了前面这些铺垫，现在我们就可以轻松地开始看`try_add_one`的实现了:
```python
    def try_add_one(self, pending_req: PendingReq) -> Req | None:
        if self.token_budget <= 0:
            return None

        if chunked_req := pending_req.chunked_req:
            # 特殊情况，先不看了

        if resource := self._try_allocate_one(pending_req):
        cache_handle, table_idx = resource
        return self._add_one_req(
            pending_req=pending_req,
            cache_handle=cache_handle,
            table_idx=table_idx,
            cached_len=cache_handle.cached_len,
        )

        return None
```

可以看到，核心逻辑是调用了`_try_allocate_one`得到了一个resource，然后用这个resource作为参数调用
`_add_one_req`，并将其结果返回。因此，接下来剖析一下这两个函数

#### PrefillAdder._try_allocate_one
```python
class PrefillAdder:
    token_budget: int
    reserved_size: int
    cache_manager: CacheManager
    table_manager: TableManager

    def _try_allocate_one(self, req: PendingReq) -> Tuple[BaseCacheHandle, int] | None:
        if self.table_manager.available_size == 0:
            return None

        # TODO: consider host cache match case
        handle = self.cache_manager.match_req(req).cuda_handle
        cached_len = handle.cached_len # 目前 RadixPrefixCache 和 req.input_ids的最长相同前缀的长度
        # TODO: better estimate policy
        extend_len = req.input_len - cached_len # 需要通过 prefill 计算KVCache的token数
        estimated_len = extend_len + req.output_len 
        # req.ouptut_len 其实是一个函数，只是@property使得它的访问方式很像一个成员变量，代码如下:
        # 而 sampling_params 来自于用户请求， max_tokens的值默认是1024, 意为本次推理最多生成的输出 Token 数
        #    @property
        #    def output_len(self) -> int:
        #        return self.sampling_params.max_tokens

        if estimated_len + self.reserved_size > self.cache_manager.available_size:
            return None

        self.cache_manager.lock(handle)
        if estimated_len + self.reserved_size > self.cache_manager.available_size:
            return self.cache_manager.unlock(handle)

        table_idx = self.table_manager.allocate()
        if cached_len > 0:  # NOTE: set the cached part
            device_ids = self.table_manager.token_pool[table_idx][:cached_len]
            page_entry = self.table_manager.page_table[table_idx][:cached_len]
            device_ids.copy_(req.input_ids[:cached_len].pin_memory(), non_blocking=True)
            page_entry.copy_(handle.get_matched_indices())

        return handle, table_idx
```

1.调用 `cache_manager.match_req(req)`，这是从 `RadixPrefixKVCache` 中查询 KVCache，`cached_len` 就是 KVCache 对应的Token长度
2.估计了一下这个 `PendingReq` 完成推理后大概要占多少个token，如果会超过 cache 的容量，则直接返回 None
3.调用 `cache_manager.lock(handle)`，增加基树上相关节点的引用计数，防止被缓存驱逐
4.调用 `table_manager.allocate()`，为这个`PendingReq` 在 `table_manager` 中分配一个位置
5.往 `table_manager` 的 `token_pool`(保存的是token id) 和 `page_table`(保存的是 KVCache的Slot id) 中填入对应的缓存数据
6.返回KVCache 和当前 PendingReq 在 table_manager 中的索引

由此可见，所谓的 `_try_allocate_one` 在 `token_pool` 中为这个`PendingReq`分配了一行，并且讲`PendingReq`命中KVCache的token写入了这个行，
除此之外，它还往 `page_table` 中填入了KVCache对应的SlotId

#### PrefillAdder._add_one_req

`try_add_one`有两个核心的函数，`_try_allocate_one`和`_add_one_req`，我们前面分析了第一个函数，接下来学习第二个函数

```python
    def _add_one_req(
        self,
        pending_req: PendingReq,
        # 下面这3参数是 __try_allocate_one 返回的
        cache_handle: BaseCacheHandle,
        table_idx: int,
        cached_len: int,
    ) -> Req:
        remain_len = pending_req.input_len - cached_len # 要通过 Prefill 计算KVCache的token数量
        chunk_size = min(self.token_budget, remain_len)
        # is_chunked = chunk_size < remain_len        # 我们先不关注ChunkedPrefil相关的内容，因此无视这两行代码
        # CLS = ChunkedReq if is_chunked else Req
        self.token_budget -= chunk_size # token_budget 是 Scheduler 的配置变量，默认值是8192
        self.reserved_size += remain_len + pending_req.output_len # 默认值是1024
        # NOTE: update the tokens ids only; new pages will be allocated in the scheduler

        # 可以把 用户的 input_ids 按照有无KVCache 分为有KVCache 的前半段，以及无KVCache 的后半段
        # 这个 _slice 专门用来索引没有 KVCache 的那后半段 input_ids
        _slice = slice(cached_len, cached_len + chunk_size) 
        device_ids = self.table_manager.token_pool[table_idx, _slice]
        # pin_memory() 把 CPU Tensor 复制到一块页锁定内存，普通 CPU 内存可能被操作系统换出到磁盘，页锁定内存不会被换出, CUDA 可以通过 DMA 更高效地将数据传输到 GPU
        device_ids.copy_(pending_req.input_ids[_slice].pin_memory(), non_blocking=True)
        return CLS(
            input_ids=pending_req.input_ids[: cached_len + chunk_size], # 截取 input_ids，为了保证Prefill状态的总token数量不超过限额
            table_idx=table_idx,
            cached_len=cached_len,
            output_len=pending_req.output_len, # 默认值是1024，大概用来控制 decode 的吧？
            uid=pending_req.uid,
            cache_handle=cache_handle,
            sampling_params=pending_req.sampling_params,
        )
```

当调度器剩余的额度不足以对这个`PendingReq`进行Prefill时，就会触发`Chunk`，我们还是假设不触发`Chunk`，所以，
`__add_one_req` 会构造并返回一个 `Req` 对象，而非`ChunkedReq`对象。

总上，`__add_one_req`将 `table_manager` 的 `token_pool` 中填入没有 KVCache 的那后半段token id
最后，构造一个 `Req` 对象返回

Req对象的字段包括：token_ids, token_pool的索引，KVCache 及 KVCache对应的token长度，采样参数 等等

```python
    def schedule_next_batch(self, prefill_budget: int) -> Batch | None:
        if len(self.pending_list) == 0:
            return None
        # estimated offset due to in-flight decode

        adder = PrefillAdder(
            token_budget=prefill_budget,
            reserved_size=self.decode_manager.inflight_tokens, # inflight_tokens = 所有 decode 请求剩余可生成 token 数 + 每个请求因 page 粒度带来的保守预留
                                                                # 它主要用于 prefill 调度时做容量判断。意思是：接纳新请求前，先把已有 decode 请求未来可能还要用的
                                                                # KV cache 空间扣掉，避免新 prefill 请求把 cache 占满
            cache_manager=self.cache_manager,
            table_manager=self.table_manager,
        )
        reqs: List[Req] = []
        for pending_req in self.pending_list:
            if req := adder.try_add_one(pending_req):
                reqs.append(req)
        return Batch(reqs=reqs, phase="prefill")
```

```python
@dataclass
class BaseCaptureData:
    seq_lens: torch.Tensor # shape: [max_bs, ] 含义: batch 中每个请求当前命中 KV cache 的长度
    positions: torch.Tensor # shape: [max_bs, ] 含义: 当前正在计算的 token 在原始序列中的 position id
    cu_seqlens_k: torch.Tensor # shape: [max_bs + 1, ] cumulative sequence lengths of Key, 假设 batch 中 3 个 key 长度分别是 key_len = [2, 3, 4], 则 cu_seqlens_k = [0, 2, 5, 9]
    # 这是因为 FlashAttention 通常不会把 variable-length sequences padding 成：
        # [A A A _ _]
        # [B B B B B]
        # [C C _ _ _]
        # 而是直接压紧：
        # A A A B B B B B C C
    # 于是, cu_seqlens_k = [0, 3, 8, 10] 刚好就提供不同 sequences 之间的边界信息
    cu_seqlens_q: torch.Tensor # shape: [max_bs + 1, ]  cumulative sequence lengths of Query，含义类似上面
    page_table: torch.Tensor # [max_batch_size, max_number_of_KV_pages], 读者应该已经很熟悉它的含义了

    def create(cls, max_bs: int, max_seq_len: int, device: torch.device, **kwargs):
        # 注意了，这里的 max_seq_len 并不是一个Request的最大token数，而是一个Request所占的最大page数
        return cls(
            seq_lens=torch.ones((max_bs,), dtype=torch.int32, device=device),
            positions=torch.zeros((max_bs,), dtype=torch.int32, device=device),
            cu_seqlens_k=torch.arange(0, max_bs + 1, dtype=torch.int32, device=device),
            cu_seqlens_q=torch.arange(0, max_bs + 1, dtype=torch.int32, device=device),
            page_table=torch.zeros((max_bs, max_seq_len), dtype=torch.int32, device=device),
            **kwargs,
        )
```

#### GraphRunner与CudaGraph
`GraphRunner`是`Engine`类的一部分，它核心用途是用 `CUDA Graph` 加速大模型的 decode 阶段。
在阅读`GraphRunner`的源码之前，我们需要了解一下什么是`CUDA Graph`

##### CudaGrpah

在传统的 CUDA 执行模型下，GPU 执行任务的流程是：
1. CPU 准备数据，计算 Kernel 的 grid/block 维度、参数地址。
2. CPU 发起 Launch：通过驱动发送开销微小但累积起来非常可观的 API 指令（约几个微秒/次）。
3. GPU 接收并执行：GPU 拿到指令后开始调度算子。

在大模型推理的 Decode阶段，每一轮迭代都需要依次调用几十甚至上百个小 Kernel（如 LayerNorm、GEMM/GEMV、Self-Attention、ROPE、RMSNorm 等）。
单个 Kernel 在 GPU 上的执行时间极短，可能仅几微秒。CPU 提交 Kernel 的时间甚至长于 GPU 实际计算的时间，导致 GPU 频繁出现等待 CPU 指令的“气泡”（CPU Bound）。

为了解决问这个问题，NAVIDA研发了CUDA Graph技术，它将原先逐个 Kernel 下发的模式改为了录制-实例化的一次性提交：
1. Graph Capture：在预热阶段，通过 cudaStreamBeginCapture() 开启录制，正常运行一次模型的 Forward 流程，最后调用 cudaStreamEndCapture()。CUDA 会将这一过程涉及的所有 Kernel 节点、内存拷贝以及依赖关系记录下来，生成一张有向无环图。
2. Graph Instantiation：将捕获的图实例化(编译)为可执行的 Graph（cudaGraphExec_t），驱动会提前对图节点进行依赖分析和调度优化。
3. Graph Launch：在后续的推导循环中，CPU 只需要调用一次 cudaGraphLaunch()，整个图就会直接下发给 GPU 执行，避免了逐个算子调用的开销。

不过这一技术存在两大问题

1. 实际生产环境中的动态 Batch Size 与 
CUDA Graph 要求捕获的计算图结构、内存地址以及算子的 Grid/Block 维度是固定不变的。但生产环境的大模型推理过程中，每次推理的Batch 内的序列长度和请求数都可能是不同的
对此的解决方案是Padding，推理引擎通常会对 Batch Size 进行分桶（如预先为 Batch Size = 1, 2, 4, 8, 16 ... 各录制一张 CUDA Graph）。运行时根据当前的请求数匹配对应的 Graph。对于 Padding 带来的额外计算开销，由于省去了巨额的 CPU Launch Overhead，整体性能仍然大幅提升。

2. 动态输入指针与指针更新
每一轮 Decode 产生的 KV Cache 存储位置或输入 Tensor 的地址是会变的。如果每次都重新 Capture，录制开销反而会拉低性能。
解决方案：固定 Buffer 区域，推理引擎在初始化时会预先分配好一块固定的内存区域（如 input_tokens 指针、kv_cache 的 Block Table 缓冲区）。
每次推理前，CPU 改写这块物理地址不变的区域内的数据，而 CUDA Graph 内部依然引用相同的地址，无需更新图结构。

```python
    def _capture_graphs(self, max_seq_len: int, vocab_size: int, model: BaseLLMModel):
        self.graph_map: Dict[int, torch.cuda.CUDAGraph] = {}
        if self.max_graph_bs == 0:
            return logger.info_rank0("CUDA graph is disabled.")

        # 初始化了一个空的FACaptureData
        self.attn_backend.init_capture_graph(max_seq_len=max_seq_len, bs_list=self.graph_bs_list)

        # 阻塞 CPU 线程，等待指定 GPU 设备上的所有 CUDA Stream 任务全部执行完毕
        torch.cuda.synchronize(self.device)
        # 显式释放 PyTorch 显存分配器中当前未被引用的缓存显存，将其还给显卡驱动。因为PyTorch 为了避免频繁向
        # GPU 驱动申请/释放显存带来的巨大开销，在张量被析构后，默认不会立刻把显存还给 GPU，而是保留在缓存池里
        # 供下次直接使用。调用 empty_cache() 可以清空这些闲置的缓存
        torch.cuda.empty_cache()
        # 重置 PyTorch 内部统计的显存峰值记录（如 torch.cuda.max_memory_allocated() 的返回值）
        torch.cuda.reset_peak_memory_stats(self.device)

        logger.info_rank0(f"Start capturing CUDA graphs with sizes: {self.graph_bs_list}")
        free_memory = get_free_memory(self.device) 
        logger.info_rank0(f"Free GPU memory before capturing CUDA graphs: {mem_GB(free_memory)}")

        self.buffer = GraphCaptureBuffer.init(self.max_graph_bs, vocab_size, self.device)

        # tqdm，python中的进度条库
        pbar = tqdm(
            sorted(self.graph_bs_list, reverse=True),
            desc="Preparing for capturing CUDA graphs...",
            unit="batch",
            disable=not get_tp_info().is_primary(),  # disable for non-primary ranks
        )

        pool = None
        for bs in pbar:
            free_memory = get_free_memory(self.device)
            pbar.desc = f"Capturing graphs: bs = {bs:<3} | avail_mem = {mem_GB(free_memory)}"
            pbar.refresh()
            graph = torch.cuda.CUDAGraph()
            batch = Batch(reqs=[self.dummy_req] * bs, phase="decode")
            batch.padded_reqs = batch.reqs
            self.attn_backend.prepare_for_capture(batch)
            self.buffer.set_batch(batch)
            with get_global_ctx().forward_batch(batch):
                self.buffer.logits[:bs] = model.forward()
                with torch.cuda.graph(graph, pool=pool, stream=self.stream):
                    self.buffer.logits[:bs] = model.forward()
            if pool is None:
                pool = graph.pool()  # reuse cuda graph handle to reduce memory
            self.graph_map[bs] = graph

        free_memory = get_free_memory(self.device)
        logger.info_rank0(f"Free GPU memory after capturing CUDA graphs: {mem_GB(free_memory)}")
```

这个类本质上是 CUDA Graph replay 时用的一组固定地址 GPU 缓冲区，因为 CUDA Graph 的一个关键要求：

```python
class GraphCaptureBuffer:
    input_ids: torch.Tensor # shape [batch_size, ] 当前 decode step 每个 request 要输入的 token id
    out_loc: torch.Tensor # shape [batch_size, ] 当前 token 对应的 KV Cache 写入位置
    positions: torch.Tensor # shape [batch_size, ] 当前 token 在各自序列中的 position，用于 RoPE 等
    logits: torch.Tensor # shape [batch_size, vocab_size] 模型 forward 的最终输出 logits

    @classmethod
    def init(cls, bs: int, vocab_size: int, device: torch.device) -> GraphCaptureBuffer:
        return GraphCaptureBuffer(
            input_ids=torch.zeros(bs, dtype=torch.int32, device=device),
            out_loc=torch.zeros(bs, dtype=torch.int32, device=device),
            positions=torch.zeros(bs, dtype=torch.int32, device=device),
            logits=torch.empty(bs, vocab_size, dtype=torch.float32, device=device),
        )

    def set_batch(self, batch: Batch) -> None:
        _slice = slice(batch.padded_size) # padded_size = len(batch.padded_reqs),  
        batch.input_ids = self.input_ids[_slice]  # [_slice] 等价于 [: len(batch_padded_reqs)]
        batch.out_loc = self.out_loc[_slice]
        batch.positions = self.positions[_slice]
```

第三:
```python
    batch = Batch(reqs=[self.dummy_req] * bs, phase="decode")
    batch.padded_reqs = batch.reqs
    self.attn_backend.prepare_for_capture(batch)
    self.buffer.set_batch(batch)
```
创建了一个由 `dummy_req` 组成的 decode阶段的`batch`，然后调用了 `attn_backend.prepare_for_caputure` 来给 Batch 设置元数据
（元数据实际上指向了那个空的 CaptureData），随后调用了 buffer.set_batch，将batch的几个成员变量指向 GraphCaptureBuffer

第四: 
进行捕捉
```python

        self.buffer.logits[:bs] = model.forward()
        with torch.cuda.graph(graph, pool=pool, stream=self.stream):
            self.buffer.logits[:bs] = model.forward()
```

`with get_global_ctx().forward_batch(batch)` 其实是把 batch 设置到全局上下文里面，然后在 model.forward() 里面读出来，输入给模型

这里有个有意思的点，为什么要进行2次forward? 第一次forward是预热，第二次才是真正的录入
第一次forward把不适合录进`Cuda Graph`的一次性工作提前做掉,比如 CUDA kernel/module lazy loading, PyTorch allocator 第一次申请某些 buffer 等等

综上，在GraphRunner的构造函数中，对GPU上的一些状态进行了清空，然后通过多次推理，录制了不同 batchSize 情形下的 `Cuda Graph`

###### GraphRunner.pad_batch
有了前面的知识，我们知道，`CudaGraph` 对decode加速的一个基本前提，就是 batch_size 必须取特定的值（在本项目中是 1, 2, 4, 8, 16, 等等)，
而 pad_batch 的作用就是将运行时Batch的 reqs 数目补白到对应的大小
```python
class GraphRunner:
    def can_use_cuda_graph(self, batch: Batch) -> bool:
        return batch.is_decode and batch.size <= self.max_graph_bs

    def pad_batch(self, batch: Batch) -> None:
        padded_size = (  # choose the first available batch size
            next(bs for bs in self.graph_bs_list if bs >= batch.size)
            if self.can_use_cuda_graph(batch)
            else batch.size
        )
        batch.padded_reqs = batch.reqs + [self.dummy_req] * (padded_size - batch.size)
```

```python
def _make_positions(batch: Batch, device: torch.device) -> torch.Tensor:
    needed_size = sum(r.extend_len for r in batch.padded_reqs)
    indices_host = torch.empty(needed_size, dtype=torch.int32, pin_memory=True)
    offset = 0
    for req in batch.padded_reqs:
        length = req.extend_len
        torch.arange(
            req.cached_len,
            req.device_len,
            dtype=torch.int32,
            out=indices_host[offset : offset + length],
        )
        offset += length
    return indices_host.to(device, non_blocking=True)
```

```python
    @property
    def extend_len(self) -> int:
        # 在 Req.__post_init__ 函数中， device_len 被赋值为 len(self.input_ids)
        return self.device_len - self.cached_len
```

`_make_input_tuple`返回一个二元组，元祖中的第一个 `mapping_host` 就是每个token在token_pool中的行号，
而positions就是每个token在token_pool里的列号。可以使用这个二元祖来索引`TableManager`的`token_pool`或

```python
def _make_input_tuple(batch: Batch, device: torch.device) -> Indice2D:
    mapping_host = torch.empty(len(batch.positions), dtype=torch.int64, pin_memory=True)
    offset = 0
    for req in batch.padded_reqs:
        length = req.extend_len
        mapping_host[offset : offset + length].fill_(req.table_idx)
        offset += length
    return mapping_host.to(device, non_blocking=True), batch.positions.to(torch.int64)
```

```python
def _make_write_tuple(batch: Batch, device: torch.device) -> Indice2D:
    mapping_list = [req.table_idx for req in batch.reqs]
    mapping_host = torch.tensor(mapping_list, dtype=torch.int64, pin_memory=True)
    write_list = [(req.device_len if req.can_decode else -1) for req in batch.reqs]
    write_host = torch.tensor(write_list, dtype=torch.int64, pin_memory=True)
    return mapping_host.to(device, non_blocking=True), write_host.to(device, non_blocking=True)
```

```python
class FlashAttentionBackend(BaseAttnBackend):
    def prepare_metadata(self, batch: Batch) -> None:
        reqs = batch.padded_reqs

        padded_size = len(reqs) # 取值一般是 [1, 2, 4, 8, 16, 24 ...] 中的某个
        seqlens_q = [req.extend_len for req in reqs] # 每个请求本轮需要计算多少个 token id
        seqlens_k = [req.device_len for req in reqs] # 每个请求各自的 input_ids 数量
        cached_lens = [req.cached_len for req in reqs] # 每个请求命中缓存的token数量
        max_seqlen_k = max(seqlens_k)
        max_seqlen_q = max(seqlens_q)
        CPU_KWARGS = {"device": "cpu", "dtype": torch.int32, "pin_memory": True}

        device = self.kvcache.device
        # cache_seqlens 表示当前执行 Attention 时，每个请求在 KV Cache 的所有有效的 slot id
        # 注意，此时[cached_len: device_len] 这几个slot对应的KVCache还没算出来，不过这些 KVCache 的地址已经被分配了
        # 标记了做FalshAttention时，每个请求的 KV Cache 应该读到哪里为止
        cache_seqlens = torch.tensor(seqlens_k, **CPU_KWARGS)
        cache_seqlens = cache_seqlens.to(device, non_blocking=True)
        # cu 即 cumlative， 前面已经介绍过了，这是因为flash attention 会把batch给压平，所以需要 cu_*** 来标记各个 Req 的边界
        cu_seqlens_k = torch.tensor([0] + seqlens_k, **CPU_KWARGS).cumsum_(dim=0)
        cu_seqlens_k = cu_seqlens_k.to(device, non_blocking=True)

        if max_seqlen_q == 1: 
            # 说明这是 decode 阶段，seqlens_q 一定是 [1, 1, 1, ...]，因此 cu_seqlens_q 一定是 [0, 1, 2, 3, ...]
            cu_seqlens_q = torch.arange(0, padded_size + 1, device=device, dtype=torch.int32)
        elif all(l == 0 for l in cached_lens):  
            # 没有任何缓存命中的 prefill 阶段，这意味着要计算所有的Q
            cu_seqlens_q = cu_seqlens_k
        else:  
            # 部分Cache命中的 Prefill 阶段(比如，一个 seq 有 10 token，前5个命中了缓存)
            cu_seqlens_q = torch.tensor([0] + seqlens_q, **CPU_KWARGS).cumsum_(dim=0)
            cu_seqlens_q = cu_seqlens_q.to(self.kvcache.device, non_blocking=True)

        page_table = get_global_ctx().page_table # 注意，虽然名字是 page_table，但它存储的其实是 slot id，而不是 page id

        # TODO: 为什么取每隔 page_size 个 slot id 除 page_size，就能得到 page_id 呢？
        # 感觉想要满足这个条件的话，那么 req 里的 cache, seqlen 等都应该是 page size 的整数倍，
        # max_seqlen_k 应该也是 page size的整数倍, 可以去看看能否查到这样的约束
        new_page_table = torch.stack(  # NOTE: global page table treat page_size = 1, we need slice
            [page_table[req.table_idx, : max_seqlen_k : self.page_size] for req in reqs]
        ) # shape: [batch_size, max_seq_len_k // page_size]，每个元素都是一个 page 起始位置的 slot id

        if self.page_size > 1:
            new_page_table.div_(self.page_size, rounding_mode="floor") # slod id -> page id

        batch.attn_metadata = FAMetadata(
            cu_seqlens_k=cu_seqlens_k,
            cu_seqlens_q=cu_seqlens_q,
            cache_seqlens=cache_seqlens,
            max_seqlen_k=max_seqlen_k,
            max_seqlen_q=max_seqlen_q,
            page_table=new_page_table,
        )
```

##### 总结：
前面我们详细分析了6个步骤，简短总结一下发生了什么
首先，对batch进行了补白，将它的batchSize变为CudaGraph要求的大小，
然后为新的即将被计算出来的KVCache分配槽位，（如果槽位不够，则对`RadixPrefixCache`执行驱逐）
再然后，进行了大量的坐标计算，通过这些坐标我们可以快速索引到这个batch中的 Seq 对应的KVCache 等信息
最后，给Batch添加上每个AttentionBackend各自需要的元数据
完成上面的几步后，将 batch 和坐标等数据打包成一个 ForwardInput 返回

### Scheduler._forward


最后计算了`AttentionBackend`执行推理所需的坐标信息 完成上面的几步后，将 batch 和 坐标信息打包成一个 `ForwardInput` 返回了

#### Scheduler._forward 源码初探
```python
class Scheduler(SchedulerIOMixin):
    def _forward(self, forward_input: ForwardInput) -> ForwardOutput:
        batch, sample_args, input_mapping, output_mapping = forward_input
        batch.input_ids = self.token_pool[input_mapping]
        forward_output = self.engine.forward_batch(batch, sample_args)
        self.token_pool[output_mapping] = forward_output.next_tokens_gpu
        self.decode_manager.filter_reqs(forward_input.batch.reqs)
        return forward_output
```

这个函数首先使用了`input_mapping`从`token_pool`索引出了所有的`input_ids`（注意，得到的`batch.input_ids`是一维的了，这里隐式进行了一次展平操作）

细心的读者可能注意到了: 这个`self.token_pool`实际上是`Scheduler.token_pool`，我们前面接触的却是`Table_manager.token_pool`，二者有何关系？
其实这两个本就是同一个`token_pool`，见 `Scheduler`的构造函数:

#### Engine.forward_batch
```python
@dataclass
class Context:
    @contextmanager
    def forward_batch(self, batch: Batch):
        assert self._batch is None, "Nested forward_batch is not allowed"
        try:
            self._batch = batch
            yield
        finally:
            self._batch = None

class GraphRunner:
    def can_use_cuda_graph(self, batch: Batch) -> bool:
        return batch.is_decode and batch.size <= self.max_graph_bs

class Engine:
    def forward_batch(self, batch: Batch, args: BatchSamplingArgs) -> ForwardOutput:
        assert torch.cuda.current_stream() == self.stream
        with self.ctx.forward_batch(batch):
            if self.graph_runner.can_use_cuda_graph(batch):
                logits = self.graph_runner.replay(batch)
            else:
                logits = self.model.forward()

        for req in batch.reqs:
            req.complete_one()

        # next_tokens_gpu 只有在 decode阶段才真正有意义。在decode阶段，每个req在每轮推理时，各自把自己的最后一个token放入 batch.inputs_ids
        # 而 len(logits)又等于 len(batch.input_ids)，所以此时有 len(batch.reqs) == len( batch.input_ids ) == len(logits), 
        # batch.size 的值实际上就是 len(batch.reqs)
        # 但当进行 Prefill时，每个req会把多个 token 放入 batch.inputs，此时 len(batch.reqs) < len(batch.input_ids),
        # next_tokens_gpu 自然就没有任何意义了? 嗯？结合下文看来，似乎未必。。。 TODO: 搞清楚，究竟怎么一回事？
        next_tokens_gpu = self.sampler.sample(logits[: batch.size], args).to(torch.int32)
        next_tokens_cpu = next_tokens_gpu.to("cpu", non_blocking=True)
        copy_done_event = torch.cuda.Event()
        copy_done_event.record(self.stream)
        return ForwardOutput(next_tokens_gpu, next_tokens_cpu, copy_done_event)
```

由上述代码可知，在Prefill阶段，会通过 `logits = self.model.forward()` 来获得  `logits`笔者以Qwen3_moe模型为例，
读了一下 `model.forward()` 的细节，发现大有门道，后面会单独开一篇来讲。我们暂且忽略 model.forward 的推理细节，接着往下看。

在完成了推理之后，对 batch 中的所有 Req 对象调用了 req.complete_one()
```python
class Req:

    def __post_init__(self) -> None:
        # ...
        self.device_len = len(self.input_ids)
        # ...

    def complete_one(self) -> None:
        self.cached_len = self.device_len
        self.device_len += 1
```
这个函数只是更新了一下可用KVCache的长度，并且递增了 device_len(Req的token序列的长度)

再然后，通过`self.sampler.sample`来对logits采样来获得新的token，代码如下，sample_impl 这个函数
是对 flashinfer 的采样器的封装，它的实现细节我们就不关注啦

```python
class Sampler:
    @nvtx_annotate("Sampler") # 用于性能观测，不影响调用结果
    def sample(self, logits: torch.Tensor, args: BatchSamplingArgs) -> torch.Tensor:
        with torch.cuda.nvtx.range("Sampler"):
            if args.temperatures is None:  # greedy sampling
                return torch.argmax(logits, dim=-1)
            return sample_impl(logits.float(), args.temperatures, args.top_k, args.top_p)
```


最后，一个值的注意的细节是

这是由于前面的发起了一个异步的 next_tokens_gpu 到 next_tokens_cpu 拷贝。因为 non_blocking=True，Python 不会等待数据复制完成，就会继续执行。后续 scheduler 在读取 CPU 结果前会调用copy_done.synchronize() 来确保此次拷贝顺利完成

```python
    def _process_last_data(self, last_data: ForwardData | None) -> None:
        if last_data is None:
            return

        batch, (_, next_tokens_cpu, copy_done) = last_data[0].batch, last_data[1]
        copy_done.synchronize()  # 前面有讲过，这是为了确保gpu上的 next_tokens_gpu 确实被拷贝到了 next_tokens_cpu 中
        reply: List[DetokenizeMsg] = []
        new_finished_reqs: Set[Req] = set()
        with self.cache_manager.lazy_free_region():
            for i, req in enumerate(batch.reqs):
                if isinstance(req, ChunkedReq): # 为何 Chunkedreq 就可以跳过？
                    continue
                next_token = next_tokens_cpu[i]
                req.append_host(next_token.unsqueeze(0))
                next_token = int(next_token.item())
                finished = not req.can_decode # 如果是 Prefill 阶段，那么推理一次就可以停止了
                if not req.sampling_params.ignore_eos: # 或者如果 decode 阶段推理得到了 eos, 也可以停止
                    finished |= next_token == self.eos_token_id
                # 保存推理得到的 next_token，返回给 Msg 的发送者
                reply.append(DetokenizeMsg(uid=req.uid, next_token=next_token, finished=finished))

                # NOTE: overlap scheduling may make the request freed twice, skip second free
                if finished and req not in self.finished_reqs: # 为何要持有 finished_reqs? 是为了防止 overlap_loop 出错
                    self.decode_manager.remove_req(req)
                    self._free_req_resources(req)
                    new_finished_reqs.add(req)
                elif batch.is_prefill:  # for prefill, non-chunk req, cache the prefix
                    self.cache_manager.cache_req(req, finished=False)

        self.finished_reqs = new_finished_reqs
        self.send_result(reply)
```

##### CacheManager.lazy_free_region()
with `cache_manager.lazy_free_region`这个函数很短，它需要配合 with 上下文管理器来使用，它的的作用就是
在with上下文结束时，惰性释放 slot_id。你知道为什么 lazy_free要做 `indices[:: self.page_size]`吗？
想想看，`TableManager` 和 `CacheManger`好像都持有 `free_slots` 和 `page_table`，那么`TableManager.free_slots` 
和`CacheManger.free_slots` 意义相同吗？ `TableManager.page_table` 和 `CacheManger.page_table`意义相同吗？

```python
    @contextmanager
    def lazy_free_region(self):
        def lazy_free(indices: torch.Tensor) -> None:
            lazy_free_list.append(indices[:: self.page_size])

        lazy_free_list: List[torch.Tensor] = []
        try:
            self._free = lazy_free
            yield
        finally:
            del self._free
            self.free_slots = torch.cat([self.free_slots] + lazy_free_list)
```

##### Req.append_host
在 with 内，遍历了batch中的每个Req，对于每个Req对应的 next_token， 都调用了 req.append_host
该函数仅仅将 next_token 添加到 input_ids 的末尾
```python
class Req:
    def append_host(self, next_token: torch.Tensor) -> None:
        self.input_ids = torch.cat([self.input_ids, next_token])
```

调用完 append_host 后，直接将此次推理产生的 next_token 放到 reply 里，后续会将 reply 返回给调用者

##### 存储KVCache
存储 KVCache 的逻辑其实有一个分支，这个分支判断时用到了 finished_reqs 这个变量，这个变量里存储的是上一轮
`_process_last_data`时被标记为完成的Req

```python
    # NOTE: overlap scheduling may make the request freed twice, skip second free
    if finished and req not in self.finished_reqs: 
        self.decode_manager.remove_req(req)
        self._free_req_resources(req)
        new_finished_reqs.add(req)
    elif batch.is_prefill:  # for prefill, non-chunk req, cache the prefix
        self.cache_manager.cache_req(req, finished=False)
```
仔细看看，这个分支的判断条件其实有点怪: 所有的 prefill 阶段都必定将 finished 设置为 true，而根据前面我们讲解的调度逻辑，
也不会把同一个`Req`调度两次，因此它不可能存在于 `finished_reqs` 里，所以这个 if 判断是一定会成立的，那么为啥会有一个 
elif batch.is_prefill 呢？
其实这段逻辑是为 `overlap loop` 准备的，后文我们会讲

上面这段逻辑中，最重要的函数就是 `self._free_req_resources` 和 `self.cache_manager.cache_req`
，而`self._free_req_resources`的逻辑是:
```python
    def _free_req_resources(self, req: Req) -> None:
        self.table_manager.free(req.table_idx)
        self.cache_manager.cache_req(req, finished=True)
```
原来 `_free_req_resources` 只是在 `self.cache_manager.cache_req` 的基础上释放了 `table_manager` 中资源，
`cache_manager.cache_req` 才是最核心的函数,而它的原理，我们已经在前面讲过了，忘记了的读者不妨重新读一遍
