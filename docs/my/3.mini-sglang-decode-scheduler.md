# DecoderManager

```python
@dataclass
class DecodeManager:
    page_size: int
    running_reqs: Set[Req] = field(default_factory=set)

    def filter_reqs(self, reqs: Iterable[Req]) -> None:
        self.running_reqs = {req for req in self.running_reqs.union(reqs) if req.can_decode}

    def remove_req(self, req: Req) -> None:
        self.running_reqs.discard(req)

    def abort_req(self, uid: int) -> Req | None: # 当推理引擎收到了 `AbortBackendMsg`时，就会调用这个
        for req in self.running_reqs:
            if req.uid == uid:
                self.running_reqs.remove(req)
                return req
        return None

    @property
    def inflight_tokens(self) -> int:
        tokens_reserved = (self.page_size - 1) * len(self.running_reqs)  # 1 page reserved
        return sum(req.remain_len for req in self.running_reqs) + tokens_reserved

    def schedule_next_batch(self) -> Batch | None:
        if not self.runnable:
            return None
        return Batch(reqs=sorted(self.running_reqs, key=lambda req: req.uid), phase="decode")

    @property
    def runnable(self) -> bool: # 6, 看来只要 running_reqs 不为空就是可运行的
        return len(self.running_reqs) > 0
```

出人意料的是， `DecoderManager` 非常地简短，仅仅几十行。上面的函数本身无需解释大家都能看懂，我们要考虑的是，`DecoderManager`是如何参与整个推理流程的

仔细看`inflight_tokens`的内容:
```python
@dataclass
class DecodeManager:
    @property
    def inflight_tokens(self) -> int:
        tokens_reserved = (self.page_size - 1) * len(self.running_reqs)  # 1 page reserved
        return sum(req.remain_len for req in self.running_reqs) + tokens_reserved
```
它这里用到了 req.remain_len 字段，我们需要留意一下 req.remain_len 是怎么来的。

```python
@dataclass(eq=False)
class Req:
    input_ids: torch.Tensor  # cpu tensor
    table_idx: int
    cached_len: int
    output_len: int
    uid: int
    sampling_params: SamplingParams
    cache_handle: BaseCacheHandle

    def __post_init__(self) -> None:
        assert self.input_ids.is_cpu
        self.device_len = len(self.input_ids)
        self.max_device_len = len(self.input_ids) + self.output_len
        assert 0 <= self.cached_len < self.device_len <= self.max_device_len

    @property
    def remain_len(self) -> int:
        return self.max_device_len - self.device_len
```

如上，`remain_len` 和 `max_device_len` 有关，而 `max_device_len` 又和 `output_len`有关。那么，`output_len`又是啥呢？

我们在 `PrefillManager`的`_add_one_req` 里找到了线索：
```python
    def _add_one_req(

        pending_req: PendingReq,

            input_ids=pending_req.input_ids[: cached_len + chunk_size],
            table_idx=table_idx,
            cached_len=cached_len,

            uid=pending_req.uid,
            cache_handle=cache_handle,

```
线索指向了 `PendingReq`:
```python
@dataclass
class PendingReq:
    uid: int
    input_ids: torch.Tensor
    sampling_params: SamplingParams
    chunked_req: ChunkedReq | None = None

    @property
    def input_len(self) -> int:
        return len(self.input_ids)

    @property
    def output_len(self) -> int: # 在这里！
        return self.sampling_params.max_tokens
```
至此，我们知道了，output_len 其实就是请求中的 max_tokens，进一步可知，`remain_len` 就是`max_tokens`

这个`inflight_tokens`是一种估算方法，估计的是一个已经进入 decode 阶段的请求，它**至多**还会额外占用多少个 token，由于这个推理引擎
是以页为粒度管理的，所以需要加上一个 page_size - 1 来防止估算值低于实际占用量

## 第四个相关点: Scheduler._process_last_data
```python
class Scheduler(SchedulerIOMixin):
    def _process_last_data(self, last_data: ForwardData | None) -> None:
        if last_data is None:
            return

        batch, (_, next_tokens_cpu, copy_done) = last_data[0].batch, last_data[1]
        copy_done.synchronize()
        reply: List[DetokenizeMsg] = []
        new_finished_reqs: Set[Req] = set()
        with self.cache_manager.lazy_free_region():
            for i, req in enumerate(batch.reqs):
                if isinstance(req, ChunkedReq):
                    continue
                next_token = next_tokens_cpu[i]
                req.append_host(next_token.unsqueeze(0))
                next_token = int(next_token.item())
                finished = not req.can_decode
                if not req.sampling_params.ignore_eos:
                    finished |= next_token == self.eos_token_id
                reply.append(DetokenizeMsg(uid=req.uid, next_token=next_token, finished=finished))

                # NOTE: overlap scheduling may make the request freed twice, skip second free
                if finished and req not in self.finished_reqs:
                    self.decode_manager.remove_req(req) # 在这里！
                    self._free_req_resources(req)
                    new_finished_reqs.add(req)
                elif batch.is_prefill:  # for prefill, non-chunk req, cache the prefix
                    self.cache_manager.cache_req(req, finished=False)

        self.finished_reqs = new_finished_reqs
        self.send_result(reply)
```

这里的调用就很简单了，移除已经完成了的 Req
